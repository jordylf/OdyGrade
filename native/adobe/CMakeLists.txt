if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Initial Adobe adapter build requires Windows x64/MSVC")
endif()
if(NOT EXISTS "${ODYGRADE_ADOBE_SDK}/Examples/Headers/AE_Effect.h" OR NOT EXISTS "${ODYGRADE_ADOBE_SDK}/Examples/Resources/PiPLtool.exe")
 message(FATAL_ERROR "Provide the extracted real Adobe C++ SDK root via ODYGRADE_ADOBE_SDK")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(SDK "${ODYGRADE_ADOBE_SDK}/Examples")
set(PIPL "${CMAKE_CURRENT_BINARY_DIR}/OdyGradePiPL.rc")
add_custom_command(OUTPUT "${PIPL}"
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/build-pipl.py" "${CMAKE_CXX_COMPILER}" "${ODYGRADE_ADOBE_SDK}" "${CMAKE_CURRENT_SOURCE_DIR}/OdyGradePiPL.r" "${PIPL}"
 DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/OdyGradePiPL.r" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/build-pipl.py"
 VERBATIM)
add_library(OdyGrade MODULE adapter.cpp "${PIPL}")
target_include_directories(OdyGrade PRIVATE "${SDK}/Headers" "${SDK}/Headers/SP" "${SDK}/Headers/Win" "${SDK}/Util" "${SDK}/Resources")
target_compile_definitions(OdyGrade PRIVATE MSWindows WIN32 _WINDOWS)
target_compile_options(OdyGrade PRIVATE /W4 /fp:precise /EHsc /utf-8)
target_link_libraries(OdyGrade PRIVATE odygrade-core comdlg32)
set_target_properties(OdyGrade PROPERTIES PREFIX "" SUFFIX ".aex")
add_executable(odygrade-adobe-tests harness.cpp adapter.cpp)
target_include_directories(odygrade-adobe-tests PRIVATE "${SDK}/Headers" "${SDK}/Headers/SP" "${SDK}/Headers/Win" "${SDK}/Util" "${SDK}/Resources")
target_compile_definitions(odygrade-adobe-tests PRIVATE MSWindows WIN32 _WINDOWS)
target_compile_options(odygrade-adobe-tests PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(odygrade-adobe-tests PRIVATE odygrade-core comdlg32)
